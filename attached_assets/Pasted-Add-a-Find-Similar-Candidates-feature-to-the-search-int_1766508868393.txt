6. Update handleSearch function to handle both modes:

const handleSearch = async () => {
  if (!query.trim()) return;
  
  setLoading(true);
  setSearchPerformed(true);
  
  try {
    const endpoint = searchMode === 'url' ? '/api/find-similar' : '/api/search';
    
    const response = await fetch(endpoint, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ 
        query: searchMode === 'url' ? query.trim() : query,
        mode: searchMode 
      })
    });
    
    const data = await response.json();
    
    if (searchMode === 'url' && data.sourceProfile) {
      setSourceProfile(data.sourceProfile); // Display who we're finding similar to
    }
    
    setResults(data.results || []);
  } catch (error) {
    console.error('Search error:', error);
  } finally {
    setLoading(false);
  }
};
